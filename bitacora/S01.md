  **Código inicial**

La primera versión de código presentada considera la lectura analógica del sensor, su conversión a un valor de índice UV y la comunicación del resultado mediante la pantalla, el LED RGB y señales sonoras.


#include <TM1637Display.h>

// Pines del Display
#define CLK 4
#define DIO 5
TM1637Display display(CLK, DIO);

// Pines de componentes
const int uvPin = A0;
const int botonPin = 2;
const int buzzerPin = 3;

// Pines del LED RGB (Pines PWM para mezclar colores)
const int ledRojo = 9;
const int ledVerde = 10;
const int ledAzul = 11;

void setup() {
  // Configuramos el botón con resistencia PULLUP interna (simplifica el cableado)
  pinMode(botonPin, INPUT_PULLUP);
  pinMode(buzzerPin, OUTPUT);
  pinMode(ledRojo, OUTPUT);
  pinMode(ledVerde, OUTPUT);
  pinMode(ledAzul, OUTPUT);
  
  // Iniciar display al máximo de brillo (0x0f)
  display.setBrightness(0x0f);
  display.clear(); // Mostrar apagado al inicio
  
  apagarLED();
}

void loop() {
  // Si se presiona el botón amarillo (LOW porque usamos PULLUP)
  if (digitalRead(botonPin) == LOW) {
    
    // 1. Leer el sensor GUVA-S12SD
    int lecturaAnalogica = analogRead(uvPin);
    
    // 2. Convertir lectura a voltaje (Asumiendo Arduino de 5V)
    float voltaje = lecturaAnalogica * (5.0 / 1023.0);
    
    // 3. Calcular Índice UV
    // El GUVA-S12SD produce aprox 0.1V (100mV) por cada 1 punto de Índice UV
    int indiceUV = voltaje / 0.1; 
    
    // 4. Mostrar en el display
    display.showNumberDec(indiceUV, false);
    
    // 5. Cambiar color del LED y emitir alerta si es peligroso
    actualizarAlerta(indiceUV);
    
    // Pequeña pausa para que la lectura sea estable mientras se presiona
    delay(1000); 
  } else {
    // Estado de reposo cuando no se presiona el botón
    display.clear();
    apagarLED();
  }
}

// Función para controlar el color del LED RGB y el sonido según el nivel UV
void actualizarAlerta(int uv) {
  if (uv <= 2) {
    // Bajo: Verde
    colorRGB(0, 255, 0); 
  } 
  else if (uv >= 3 && uv <= 5) {
    // Moderado: Amarillo
    colorRGB(255, 255, 0);
  } 
  else if (uv >= 6 && uv <= 7) {
    // Alto: Naranja
    colorRGB(255, 100, 0);
  } 
  else if (uv >= 8 && uv <= 10) {
    // Muy Alto: Rojo + Sonido de alerta
    colorRGB(255, 0, 0);
    tone(buzzerPin, 1000, 500); // Tono de 1kHz por 0.5 segundos
  } 
  else {
    // Extremo (11+): Morado + Sonido de alerta más agudo
    colorRGB(200, 0, 255);
    tone(buzzerPin, 1500, 800); 
  }
}

// Función auxiliar para enviar los valores al LED RGB
void colorRGB(int rojo, int verde, int azul) {
  analogWrite(ledRojo, rojo);
  analogWrite(ledVerde, verde);
  analogWrite(ledAzul, azul);
}

void apagarLED() {
  colorRGB(0, 0, 0);
}
